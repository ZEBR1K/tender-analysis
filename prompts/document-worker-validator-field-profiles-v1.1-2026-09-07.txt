function hard(message) {
  throw new Error('[Expand AI Validator units] ' + message);
}

function isObject(value) {
  return value !== null && typeof value === 'object' && !Array.isArray(value);
}

function renderList(values) {
  return values.map((value) => '- ' + value).join('\n');
}

function renderProfile(fieldKey, profile) {
  return [
    'FIELD PROFILE: ' + fieldKey,
    'TITLE: ' + profile.title,
    'DEFINITION: ' + profile.definition,
    'CONFIRM ONLY WHEN:',
    renderList(profile.confirmation_criteria),
    'SEMANTIC EXCLUSIONS:',
    renderList(profile.semantic_exclusions),
    'REJECT WHEN:',
    renderList(profile.rejected_when),
    'REQUIRES REVIEW WHEN:',
    renderList(profile.requires_review_when),
  ].join('\n');
}

const items = $input.all();
if (items.length !== 1) hard('Expected exactly one dispatch envelope.');

const canonicalSet = new Set(CANONICAL_FIELD_KEYS);
const catalogKeys = Object.keys(VALIDATOR_FIELD_PROFILES);
for (const fieldKey of CANONICAL_FIELD_KEYS) {
  if (!isObject(VALIDATOR_FIELD_PROFILES[fieldKey])) {
    hard('Missing profile for known field_key "' + fieldKey + '".');
  }
}
for (const fieldKey of catalogKeys) {
  if (!canonicalSet.has(fieldKey)) {
    hard('Unexpected profile for non-canonical field_key "' + fieldKey + '".');
  }
}
if (catalogKeys.length !== CANONICAL_FIELD_KEYS.length) {
  hard('Profile catalog cardinality mismatch: ' + catalogKeys.length + '/' + CANONICAL_FIELD_KEYS.length + '.');
}

const envelope = items[0]?.json;
if (
  envelope?.dispatch_version !== 'ai_validator_dispatch_v1' ||
  !Array.isArray(envelope.units_for_ai) ||
  envelope.units_for_ai.length === 0
) {
  hard('Invalid or empty units_for_ai.');
}

return envelope.units_for_ai.map((unit) => {
  if (!isObject(unit) || !Array.isArray(unit.verified_facts) || unit.verified_facts.length === 0) {
    hard('AI unit must contain non-empty verified_facts.');
  }

  const requested = new Set();
  for (let factIndex = 0; factIndex < unit.verified_facts.length; factIndex++) {
    const fact = unit.verified_facts[factIndex];
    const fieldKey = fact?.field_key;
    if (typeof fieldKey !== 'string' || fieldKey.length === 0) {
      hard('verified_facts[' + factIndex + '].field_key is missing.');
    }
    if (!canonicalSet.has(fieldKey)) {
      hard('Unknown field_key "' + fieldKey + '".');
    }
    if (!isObject(VALIDATOR_FIELD_PROFILES[fieldKey])) {
      hard('Missing profile for known field_key "' + fieldKey + '".');
    }
    requested.add(fieldKey);
  }

  const fieldKeys = CANONICAL_FIELD_KEYS.filter((fieldKey) => requested.has(fieldKey));
  const fieldProfilesText = fieldKeys
    .map((fieldKey) => renderProfile(fieldKey, VALIDATOR_FIELD_PROFILES[fieldKey]))
    .join('\n\n');
  const fieldProfilesByKey = Object.fromEntries(
    fieldKeys.map((fieldKey) => [fieldKey, renderProfile(fieldKey, VALIDATOR_FIELD_PROFILES[fieldKey])])
  );

  return {
    json: {
      ...unit,
      validator_field_keys: fieldKeys,
      validator_prompt_context: {
        prompt_version: VALIDATOR_PROMPT_VERSION,
        field_profiles_version: VALIDATOR_FIELD_PROFILES_VERSION,
        field_keys: fieldKeys,
        field_profiles_text: fieldProfilesText,
        field_profiles_by_key: fieldProfilesByKey,
      },
    },
    pairedItem: { item: 0 },
  };
});
