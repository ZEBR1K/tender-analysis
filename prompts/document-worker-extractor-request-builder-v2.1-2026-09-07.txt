function fail(
  itemIndex,
  message
) {
  throw new Error(
    `[Подготовить запрос для AI] Item ${
      itemIndex + 1
    }: ${message}`
  );
}


function isObject(value) {
  return (
    value !== null &&
    typeof value === 'object' &&
    !Array.isArray(value)
  );
}


function uniqueStrings(values) {
  return [
    ...new Set(
      values.filter(
        value =>
          typeof value === 'string' &&
          value.length > 0
      )
    ),
  ];
}


function arraysEqual(
  left,
  right
) {
  if (
    left.length !==
    right.length
  ) {
    return false;
  }

  for (
    let index = 0;
    index < left.length;
    index++
  ) {
    if (
      left[index] !==
      right[index]
    ) {
      return false;
    }
  }

  return true;
}


function sameStringSet(
  left,
  right
) {
  if (
    left.length !==
    right.length
  ) {
    return false;
  }

  const rightSet =
    new Set(right);

  return left.every(
    value =>
      rightSet.has(value)
  );
}


function renderFieldCatalog() {
  return FIELD_CATALOG
    .map(
      (
        field,
        index
      ) =>
        `${
          index + 1
        }. ${field.key} — ${field.title}\n` +
        `   ${field.description}`
    )
    .join('\n\n');
}


function renderAiSegment(segment) {
  return [
    `[SEMANTIC_BLOCK id="${segment.semantic_block_id}" scope="${segment.scope}" type="${segment.type}" role="${segment.role}"]`,
    segment.text,
    '[/SEMANTIC_BLOCK]',
  ].join('\n');
}

КРИТИЧЕСКОЕ ПРАВИЛО О STATUS

Если status = "found":
- review_reason_code ОБЯЗАТЕЛЬНО должен быть null;
- review_note ОБЯЗАТЕЛЬНО должен быть null;
- факт должен быть прямо и однозначно подтверждён текстом.

Если для факта требуется review_reason_code или review_note,
status ОБЯЗАТЕЛЬНО должен быть "requires_review".

Никогда не возвращай сочетание:
status = "found"
и одновременно непустой review_reason_code или review_note.


КРИТИЧЕСКОЕ ПРАВИЛО ОБ ОТСУТСТВУЮЩЕМ ЗНАЧЕНИИ

Если конкретное значение field_key прямо не указано в текущих SEMANTIC_BLOCK,
НЕ создавай fact для этого field_key.

Не используй status = "requires_review" только для сообщения,
что значение отсутствует.

requires_review используется только тогда, когда в тексте действительно
есть кандидат на значение, но сам кандидат неоднозначен, конфликтует,
неполон или повреждён.

Например:

Поле customer означает наименование организации-заказчика.

ФИО сотрудника, контрактного управляющего, специалиста по закупкам,
подписанта или контактного лица НЕ является значением customer.

Если наименование организации-заказчика отсутствует,
не возвращай customer вообще.

ФИО сотрудника может относиться к customer_contacts только если текст
действительно позволяет считать его контактным лицом заказчика.
Не превращай ФИО сотрудника в наименование customer.

18. status = "requires_review" используй, если факт релевантен, но есть существенная неоднозначность, конфликт, неполное значение, неясная ссылка или проблема качества источника.

Разрешённые review_reason_code:
- ambiguous_wording
- conflicting_evidence
- incomplete_value
- unclear_reference
- source_quality_issue
- other

19. confidence — число от 0 до 1, отражающее уверенность в том, что именно заявленный fact корректно подтверждается предоставленным текстом.

Высокий confidence не заменяет evidence.

20. value_text должен передавать содержание факта максимально близко к формулировке документа.

На этом этапе не выполняй машинную нормализацию значений.

Например:
"не менее 12 месяцев с момента подписания акта"
нужно сохранить как текст,
а не преобразовывать в объект с value=12 и unit=months.

21. Не добавляй page_no, bbox, source_block_id или другие сведения о provenance.

Они будут восстановлены системой автоматически по semantic_block_id.

22. Возвращай только JSON.
Никакого Markdown, комментариев до JSON или текста после JSON.

// 5. ОБРАБОТКА INPUT ITEMS

const items =
  $input.all();

if (
  items.length === 0
) {
  throw new Error(
    '[Подготовить запрос для AI] Нет входных items.'
  );
}


return items.map(
  (
    item,
    itemIndex
  ) => {
    const input =
      item?.json;

    if (
      !isObject(input)
    ) {
      fail(
        itemIndex,
        'Input JSON отсутствует или имеет неверный формат.'
      );
    }


    // --------------------------------------------------------
    // analysis_unit
    // --------------------------------------------------------

    const unit =
      input.analysis_unit;

    if (
      !isObject(unit)
    ) {
      fail(
        itemIndex,
        'Отсутствует analysis_unit.'
      );
    }


    const analysisUnitId =
      unit.analysis_unit_id;

    if (
      typeof analysisUnitId !== 'string' ||
      analysisUnitId.length === 0
    ) {
      fail(
        itemIndex,
        'analysis_unit.analysis_unit_id отсутствует.'
      );
    }


    if (
      !Array.isArray(
        unit.primary_semantic_block_ids
      )
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: отсутствует primary_semantic_block_ids.`
      );
    }


    if (
      !Array.isArray(
        unit.overlap_semantic_block_ids
      )
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: отсутствует overlap_semantic_block_ids.`
      );
    }


    const primaryIds =
      uniqueStrings(
        unit.primary_semantic_block_ids
      );

    const overlapIds =
      uniqueStrings(
        unit.overlap_semantic_block_ids
      );


    if (
      primaryIds.length === 0
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: нет ни одного primary semantic block.`
      );
    }


    if (
      primaryIds.length !==
      unit.primary_semantic_block_ids.length
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: primary_semantic_block_ids содержит пустые или дублирующиеся ID.`
      );
    }


    if (
      overlapIds.length !==
      unit.overlap_semantic_block_ids.length
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: overlap_semantic_block_ids содержит пустые или дублирующиеся ID.`
      );
    }


    const primarySet =
      new Set(primaryIds);

    for (
      const overlapId
      of overlapIds
    ) {
      if (
        primarySet.has(
          overlapId
        )
      ) {
        fail(
          itemIndex,
          `${analysisUnitId}: ${overlapId} одновременно primary и overlap.`
        );
      }
    }


    const expectedIds = [
      ...primaryIds,
      ...overlapIds,
    ];


    // --------------------------------------------------------
    // ai_segments
    // --------------------------------------------------------

    const aiSegments =
      input.ai_segments;

    if (
      !Array.isArray(
        aiSegments
      ) ||
      aiSegments.length === 0
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: ai_segments отсутствует или пуст.`
      );
    }


    const segmentIds =
      aiSegments.map(
        segment =>
          segment?.semantic_block_id
      );


    if (
      !arraysEqual(
        segmentIds,
        expectedIds
      )
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: порядок или состав ai_segments не совпадает с primary + overlap semantic IDs.`
      );
    }


    const uniqueSegmentIds =
      uniqueStrings(
        segmentIds
      );

    if (
      uniqueSegmentIds.length !==
      segmentIds.length
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: ai_segments содержит повторяющиеся semantic_block_id.`
      );
    }


    // --------------------------------------------------------
    // provenance
    // --------------------------------------------------------

    const provenance =
      input.provenance;

    if (
      !isObject(
        provenance
      ) ||
      !isObject(
        provenance.index
      )
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: provenance.index отсутствует.`
      );
    }


    const provenanceIndex =
      provenance.index;

    const provenanceIds =
      Object.keys(
        provenanceIndex
      );


    if (
      !sameStringSet(
        provenanceIds,
        expectedIds
      )
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: provenance.index не совпадает с semantic blocks текущей unit.`
      );
    }


    // --------------------------------------------------------
    // Валидация каждого semantic segment
    // --------------------------------------------------------

    for (
      const segment
      of aiSegments
    ) {
      const id =
        segment.semantic_block_id;

      if (
        segment.scope !== 'primary' &&
        segment.scope !== 'overlap'
      ) {
        fail(
          itemIndex,
          `${analysisUnitId}: ${id} имеет недопустимый scope "${segment.scope}".`
        );
      }


      if (
        typeof segment.text !== 'string' ||
        segment.text.trim().length === 0
      ) {
        fail(
          itemIndex,
          `${analysisUnitId}: ${id} содержит пустой text.`
        );
      }


      const provenanceEntry =
        provenanceIndex[id];

      if (
        !isObject(
          provenanceEntry
        )
      ) {
        fail(
          itemIndex,
          `${analysisUnitId}: отсутствует provenance для ${id}.`
        );
      }


      if (
        provenanceEntry.scope !==
        segment.scope
      ) {
        fail(
          itemIndex,
          `${analysisUnitId}: scope ${id} не совпадает между ai_segments и provenance.`
        );
      }
    }


    // --------------------------------------------------------
    // Дополнительная проверка счетчиков provenance
    // --------------------------------------------------------

    if (
      provenance.primary_semantic_blocks_count !==
      primaryIds.length
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: неверный primary_semantic_blocks_count в provenance.`
      );
    }


    if (
      provenance.overlap_semantic_blocks_count !==
      overlapIds.length
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: неверный overlap_semantic_blocks_count в provenance.`
      );
    }


    if (
      provenance.total_semantic_blocks_count !==
      expectedIds.length
    ) {
      fail(
        itemIndex,
        `${analysisUnitId}: неверный total_semantic_blocks_count в provenance.`
      );
    }


    // --------------------------------------------------------
    // USER PROMPT
    // --------------------------------------------------------

    const contextLines = [
      `analysis_unit_id: ${analysisUnitId}`,
    ];


    if (
      input.document?.file_name
    ) {
      contextLines.push(
        `Файл документа: ${input.document.file_name}`
      );
    }


    if (
      unit.section_title
    ) {
      contextLines.push(
        `Раздел: ${unit.section_title}`
      );
    }


    if (
      unit.part_index !== undefined &&
      unit.parts_total !== undefined
    ) {
      contextLines.push(
        `Часть раздела: ${unit.part_index} из ${unit.parts_total}`
      );
    }


    if (
      unit.context?.sheet_name
    ) {
      contextLines.push(
        `Excel-лист: ${unit.context.sheet_name}`
      );
    }


    if (
      unit.context?.sheet_title
    ) {
      contextLines.push(
        `Название Excel-листа: ${unit.context.sheet_title}`
      );
    }


    const renderedSegments =
      aiSegments
        .map(
          renderAiSegment
        )
        .join('\n\n');


    const userPrompt = `
Проанализируй только приведённую ниже часть тендерного документа.

Найди все факты из каталога tender_fields_v1, которые явно подтверждаются содержимым SEMANTIC_BLOCK.

Метаданные ниже предоставлены только для навигации и понимания контекста.
Не используй их как evidence.


КОНТЕКСТ

${contextLines.join('\n')}


SEMANTIC BLOCKS

${renderedSegments}


Верни результат строго по схеме ${CONFIG.schemaVersion}.

Если в этой analysis unit нет ни одного целевого факта, верни пустой массив facts.
`.trim();


    // --------------------------------------------------------
    // Оценка размера будущего AI-вызова
    // --------------------------------------------------------

    const systemCharacters =
      SYSTEM_PROMPT.length;

    const userCharacters =
      userPrompt.length;

    const totalCharacters =
      systemCharacters +
      userCharacters;

    const estimatedInputTokens =
      Math.ceil(
        totalCharacters /
        CONFIG.approxCharsPerToken
      );


    // --------------------------------------------------------
    // Минимальная metadata analysis unit для следующих этапов
    // --------------------------------------------------------

    const analysisUnitMeta = {
      analysis_unit_id:
        analysisUnitId,

      section_id:
        unit.section_id ??
        null,

      section_title:
        unit.section_title ??
        null,

      section_kind:
        unit.section_kind ??
        null,

      part_index:
        unit.part_index ??
        null,

      parts_total:
        unit.parts_total ??
        null,

      context:
        unit.context ??
        {},

      source_pages:
        unit.source_pages ??
        [],

      flags:
        unit.flags ??
        {},

      primary_semantic_block_ids:
        primaryIds,

      overlap_semantic_block_ids:
        overlapIds,
    };


    // --------------------------------------------------------
    // OUTPUT
    // --------------------------------------------------------

    return {
      json: {
        // Эти данные сохраняем рядом,
        // но НЕ отправляем модели целиком.
        tender:
          input.tender ??
          null,

        document:
          input.document ??
          null,

        analysis_batch:
          input.analysis_batch ??
          null,

        analysis_unit_meta:
          analysisUnitMeta,

        // Нужны после AI для проверки evidence.
        ai_segments:
          aiSegments,

        // Нужен после AI для восстановления
        // настоящего source/page/bbox.
        provenance,

        ai_request: {
          prompt_version:
            CONFIG.promptVersion,

          schema_version:
            CONFIG.schemaVersion,

          field_catalog_version:
            CONFIG.fieldCatalogVersion,

          system_prompt:
            SYSTEM_PROMPT,

          user_prompt:
            userPrompt,

          response_schema:
            RESPONSE_SCHEMA,

          field_catalog:
            FIELD_CATALOG,

          evidence_resource_contract: {
            max_evidence_per_fact:
              CONFIG.maxEvidencePerFact,
            max_quote_length:
              CONFIG.maxQuoteLength,
            max_total_quote_chars_per_fact:
              CONFIG.maxTotalQuoteCharsPerFact,
            exact_duplicates_allowed:
              false,
            primary_grounded_evidence_required:
              true,
          },

          stats: {
            system_characters:
              systemCharacters,

            user_characters:
              userCharacters,

            total_characters:
              totalCharacters,

            estimated_input_tokens:
              estimatedInputTokens,

            semantic_blocks_count:
              aiSegments.length,

            primary_blocks_count:
              primaryIds.length,

            overlap_blocks_count:
              overlapIds.length,
          },
        },
      },

      // Сохраняем связь с конкретным входным item.
      pairedItem: {
        item: itemIndex,
      },
    };
  }
);
